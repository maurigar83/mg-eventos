async function realizarLogin(usuario, password) { const MAX_INTENTOS = 2; const ESPERA_REINTENTO = 1000; for (let intento = 1; intento <= MAX_INTENTOS; intento++) { try { const response = await fetch(API_URL, { method: "POST", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify({ accion: "login", usuario: usuario, password: password }) }); const data = await response.json(); if (!data.ok) { return { ok: false, mensaje: data.mensaje || "Usuario o contraseña incorrectos." }; } return { ok: true, data: data }; } catch (error) { if (intento < MAX_INTENTOS) { await new Promise(resolve => setTimeout(resolve, ESPERA_REINTENTO) ); } } } return { ok: false, conexion: false, mensaje: "No fue posible conectar con el sistema de seguridad. Intenta nuevamente." }; }
